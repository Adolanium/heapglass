cmake_minimum_required(VERSION 3.24)
project(heapglass VERSION 0.1.0 LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Heapglass requires Windows x64 and MSVC")
endif()
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_definitions(UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
add_compile_options(/W4 /permissive- /EHsc /utf-8)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin)
set(D third_party/detours/src)
add_library(detours STATIC ${D}/detours.cpp ${D}/modules.cpp ${D}/disasm.cpp ${D}/image.cpp ${D}/creatwth.cpp ${D}/disolx86.cpp ${D}/disolx64.cpp ${D}/disolia64.cpp ${D}/disolarm.cpp ${D}/disolarm64.cpp)
target_include_directories(detours PUBLIC ${D})
target_compile_options(detours PRIVATE /W0 /permissive)
set(I third_party/imgui)
add_library(imgui STATIC ${I}/imgui.cpp ${I}/imgui_draw.cpp ${I}/imgui_tables.cpp ${I}/imgui_widgets.cpp ${I}/backends/imgui_impl_win32.cpp ${I}/backends/imgui_impl_dx11.cpp)
target_include_directories(imgui PUBLIC ${I} ${I}/backends)
target_link_libraries(imgui PUBLIC d3d11 dxgi dwmapi)
add_library(heapglass_core STATIC src/analysis.cpp src/storage.cpp)
target_include_directories(heapglass_core PUBLIC include third_party/json/single_include)
add_library(heapglass_controller STATIC src/controller.cpp)
target_link_libraries(heapglass_controller PUBLIC heapglass_core advapi32 psapi bcrypt dbghelp)
add_library(HeapglassAgent SHARED src/agent.cpp)
target_include_directories(HeapglassAgent PRIVATE include)
target_link_libraries(HeapglassAgent PRIVATE detours)
add_executable(Heapglass WIN32 src/main.cpp src/ui.cpp src/screenshot.cpp)
target_link_libraries(Heapglass PRIVATE heapglass_controller imgui comdlg32 windowscodecs ole32)
add_executable(HeapglassTarget src/target.cpp)
target_link_libraries(HeapglassTarget PRIVATE bcrypt cabinet)
add_executable(HeapglassTests tests/tests.cpp)
target_link_libraries(HeapglassTests PRIVATE heapglass_controller)
enable_testing()
add_test(NAME core COMMAND HeapglassTests)
add_test(NAME live COMMAND HeapglassTests --live)
set_tests_properties(live PROPERTIES TIMEOUT 90)
install(TARGETS Heapglass HeapglassAgent HeapglassTarget HeapglassTests RUNTIME DESTINATION .)
install(DIRECTORY third_party/ DESTINATION licenses FILES_MATCHING PATTERN "LICENSE*")
install(FILES README.md DESTINATION .)
install(FILES docs/coverage.md docs/validation.md docs/references.md docs/performance.json docs/ui-profile.json DESTINATION docs)
install(DIRECTORY docs/screenshots DESTINATION docs)
