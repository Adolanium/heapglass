cmake_minimum_required(VERSION 3.24)
project(heapglass VERSION 0.4.0 LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Heapglass requires Windows x64 and MSVC")
endif()
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_definitions(UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN IMGUI_USE_WCHAR32)
add_compile_options(/W4 /permissive- /EHsc /utf-8)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin)
set(D third_party/detours/src)
add_library(detours STATIC ${D}/detours.cpp ${D}/modules.cpp ${D}/disasm.cpp ${D}/image.cpp ${D}/creatwth.cpp ${D}/disolx86.cpp ${D}/disolx64.cpp ${D}/disolia64.cpp ${D}/disolarm.cpp ${D}/disolarm64.cpp)
target_include_directories(detours PUBLIC ${D})
target_compile_options(detours PRIVATE /W0 /permissive)
set(I third_party/imgui)
add_library(imgui STATIC ${I}/imgui.cpp ${I}/imgui_draw.cpp ${I}/imgui_tables.cpp ${I}/imgui_widgets.cpp ${I}/backends/imgui_impl_win32.cpp ${I}/backends/imgui_impl_dx11.cpp)
target_include_directories(imgui PUBLIC ${I} ${I}/backends)
target_link_libraries(imgui PUBLIC d3d11 dxgi dwmapi)
add_library(heapglass_core STATIC src/analysis.cpp src/storage.cpp src/strings.cpp src/profiling.cpp)
target_include_directories(heapglass_core PUBLIC include third_party/json/single_include)
add_library(heapglass_controller STATIC src/controller.cpp src/memory.cpp)
target_link_libraries(heapglass_controller PUBLIC heapglass_core advapi32 psapi bcrypt dbghelp)
add_library(HeapglassAgent SHARED src/agent.cpp)
target_include_directories(HeapglassAgent PRIVATE include)
target_link_libraries(HeapglassAgent PRIVATE detours)
add_library(HeapglassAgentFault SHARED src/agent.cpp)
target_include_directories(HeapglassAgentFault PRIVATE include)
target_compile_definitions(HeapglassAgentFault PRIVATE HG_VALIDATE_INSTALL_FAILURE)
target_link_libraries(HeapglassAgentFault PRIVATE detours)
add_library(HeapglassAgentLegacy SHARED tests/legacy_agent.cpp)
add_library(HeapglassAgentStackFault SHARED src/agent.cpp)
target_include_directories(HeapglassAgentStackFault PRIVATE include)
target_compile_definitions(HeapglassAgentStackFault PRIVATE HG_VALIDATE_UNSAFE_TRACE)
target_link_libraries(HeapglassAgentStackFault PRIVATE detours)
add_executable(HeapglassHandoffTests tests/handoff.cpp)
target_include_directories(HeapglassHandoffTests PRIVATE include)
add_executable(HeapglassStackTests tests/stacks.cpp)
target_link_libraries(HeapglassStackTests PRIVATE heapglass_controller)
add_executable(HeapglassFailureTests tests/failure_tests.cpp)
target_link_libraries(HeapglassFailureTests PRIVATE heapglass_controller)
add_executable(HeapglassFailureHost tests/failure_host.cpp)
target_link_libraries(HeapglassFailureHost PRIVATE heapglass_controller)
add_library(imgui_validation STATIC ${I}/imgui.cpp ${I}/imgui_draw.cpp ${I}/imgui_tables.cpp ${I}/imgui_widgets.cpp)
target_include_directories(imgui_validation PUBLIC ${I})
target_compile_definitions(imgui_validation PUBLIC IMGUI_ENABLE_TEST_ENGINE)
add_executable(HeapglassUITests tests/ui_tests.cpp src/ui.cpp src/desktop.cpp)
target_link_libraries(HeapglassUITests PRIVATE heapglass_controller imgui_validation comdlg32)
add_executable(HeapglassDialogTests tests/dialog_tests.cpp src/desktop.cpp)
target_link_libraries(HeapglassDialogTests PRIVATE heapglass_controller imgui comdlg32)
add_executable(Heapglass WIN32 src/main.cpp src/ui.cpp src/screenshot.cpp src/desktop.cpp)
target_link_libraries(Heapglass PRIVATE heapglass_controller imgui comdlg32 windowscodecs ole32)
add_executable(HeapglassTarget src/target.cpp)
target_link_libraries(HeapglassTarget PRIVATE bcrypt cabinet)
add_executable(HeapglassTests tests/tests.cpp)
target_link_libraries(HeapglassTests PRIVATE heapglass_controller)
add_executable(HeapglassStringTests tests/strings.cpp)
target_link_libraries(HeapglassStringTests PRIVATE heapglass_core)
add_executable(HeapglassRetentionTests tests/retention.cpp)
target_link_libraries(HeapglassRetentionTests PRIVATE heapglass_core)
add_executable(HeapglassProfilingTests tests/profiling.cpp)
target_link_libraries(HeapglassProfilingTests PRIVATE heapglass_core)
add_executable(HeapglassWorkflowTests tests/workflow.cpp)
target_link_libraries(HeapglassWorkflowTests PRIVATE heapglass_controller)
enable_testing()
add_test(NAME strings COMMAND HeapglassStringTests)
add_test(NAME retention COMMAND HeapglassRetentionTests)
add_test(NAME allocation-profiles COMMAND HeapglassProfilingTests)
add_test(NAME profiling-workflow COMMAND HeapglassWorkflowTests)
set_tests_properties(profiling-workflow PROPERTIES TIMEOUT 60)
add_test(NAME collector-handoff COMMAND HeapglassHandoffTests)
add_test(NAME bounded-stacks COMMAND HeapglassStackTests)
set_tests_properties(bounded-stacks PROPERTIES TIMEOUT 90)
add_test(NAME core COMMAND HeapglassTests)
add_test(NAME live COMMAND HeapglassTests --live)
set_tests_properties(live PROPERTIES TIMEOUT 90)
add_test(NAME desktop-ui COMMAND HeapglassUITests)
add_test(NAME native-dialogs-dpi COMMAND HeapglassDialogTests)
set_tests_properties(desktop-ui native-dialogs-dpi PROPERTIES TIMEOUT 90)
add_test(NAME failures COMMAND HeapglassFailureTests)
set_tests_properties(failures PROPERTIES TIMEOUT 120)
install(TARGETS Heapglass HeapglassAgent HeapglassTarget HeapglassTests RUNTIME DESTINATION .)
install(DIRECTORY third_party/ DESTINATION licenses FILES_MATCHING PATTERN "LICENSE*")
install(FILES README.md DESTINATION .)
install(FILES docs/coverage.md docs/validation.md docs/implementation.md docs/profiling.md docs/sustained-validation.md docs/sustained.json docs/references.md docs/performance.json docs/performance-before-v2.json docs/performance-v0.2.json docs/ui-profile.json DESTINATION docs)
install(DIRECTORY docs/screenshots DESTINATION docs)
